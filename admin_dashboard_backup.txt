
import 'package:flutter/material.dart';

class AdminDashboard extends StatefulWidget {
  const AdminDashboard({super.key});

  @override
  State<AdminDashboard> createState() => _AdminDashboardState();
}

class _AdminDashboardState extends State<AdminDashboard> {
  final _formKey = GlobalKey<FormState>();

  final _titleController = TextEditingController();
  final _departmentController = TextEditingController();
  final _locationController = TextEditingController();
  final _descriptionController = TextEditingController();
  final _responsibilitiesController = TextEditingController();
  final _skillsController = TextEditingController();

  String _jobType = 'Full Time';
  String _experience = '1-2 Years';
  String _education = "Bachelor's Degree";

  final Color navy = const Color(0xFF172554);
  final Color blue = const Color(0xFF4F6EF7);
  final Color background = const Color(0xFFF5F7FC);

  @override
  void dispose() {
    _titleController.dispose();
    _departmentController.dispose();
    _locationController.dispose();
    _descriptionController.dispose();
    _responsibilitiesController.dispose();
    _skillsController.dispose();
    super.dispose();
  }

  Widget inputField({
    required String label,
    required String hint,
    required TextEditingController controller,
    int maxLines = 1,
  }) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          label,
          style: const TextStyle(
            fontWeight: FontWeight.w600,
            fontSize: 13,
            color: Color(0xFF334155),
          ),
        ),
        const SizedBox(height: 9),
        TextFormField(
          controller: controller,
          maxLines: maxLines,
          validator: (value) {
            if (value == null || value.trim().isEmpty) {
              return 'Please enter $label';
            }
            return null;
          },
          decoration: InputDecoration(
            hintText: hint,
            hintStyle: const TextStyle(
              color: Color(0xFF94A3B8),
              fontSize: 13,
            ),
            filled: true,
            fillColor: const Color(0xFFFAFBFE),
            contentPadding: const EdgeInsets.all(15),
            border: OutlineInputBorder(
              borderRadius: BorderRadius.circular(10),
              borderSide: const BorderSide(color: Color(0xFFE2E8F0)),
            ),
            enabledBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(10),
              borderSide: const BorderSide(color: Color(0xFFE2E8F0)),
            ),
            focusedBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(10),
              borderSide: BorderSide(color: blue, width: 1.5),
            ),
          ),
        ),
      ],
    );
  }

  Widget dropdownField({
    required String label,
    required String value,
    required List<String> items,
    required ValueChanged<String?> onChanged,
  }) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          label,
          style: const TextStyle(
            fontWeight: FontWeight.w600,
            fontSize: 13,
            color: Color(0xFF334155),
          ),
        ),
        const SizedBox(height: 9),
        DropdownButtonFormField<String>(
          initialValue: value,
          isExpanded: true,
          items: items.map((item) {
            return DropdownMenuItem(
              value: item,
              child: Text(
                item,
                overflow: TextOverflow.ellipsis,
                style: const TextStyle(fontSize: 13),
              ),
            );
          }).toList(),
          onChanged: onChanged,
          decoration: InputDecoration(
            filled: true,
            fillColor: const Color(0xFFFAFBFE),
            contentPadding: const EdgeInsets.symmetric(
              horizontal: 14,
              vertical: 15,
            ),
            border: OutlineInputBorder(
              borderRadius: BorderRadius.circular(10),
              borderSide: const BorderSide(color: Color(0xFFE2E8F0)),
            ),
            enabledBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(10),
              borderSide: const BorderSide(color: Color(0xFFE2E8F0)),
            ),
          ),
        ),
      ],
    );
  }

  Widget sectionCard({
    required String title,
    required String subtitle,
    required IconData icon,
    required List<Widget> children,
  }) {
    return Container(
      width: double.infinity,
      padding: const EdgeInsets.all(22),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(16),
        border: Border.all(color: const Color(0xFFE9EDF5)),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            children: [
              Container(
                padding: const EdgeInsets.all(10),
                decoration: BoxDecoration(
                  color: const Color(0xFFEEF2FF),
                  borderRadius: BorderRadius.circular(10),
                ),
                child: Icon(icon, color: blue, size: 20),
              ),
              const SizedBox(width: 12),
              Expanded(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      title,
                      style: const TextStyle(
                        fontWeight: FontWeight.bold,
                        fontSize: 16,
                        color: Color(0xFF172554),
                      ),
                    ),
                    const SizedBox(height: 4),
                    Text(
                      subtitle,
                      style: const TextStyle(
                        fontSize: 12,
                        color: Color(0xFF94A3B8),
                      ),
                    ),
                  ],
                ),
              ),
            ],
          ),
          const SizedBox(height: 22),
          ...children,
        ],
      ),
    );
  }

  Widget sidebarItem(IconData icon, String title, {bool selected = false}) {
    return Container(
      margin: const EdgeInsets.only(bottom: 8),
      padding: const EdgeInsets.symmetric(horizontal: 13, vertical: 13),
      decoration: BoxDecoration(
        color: selected ? const Color(0xFF293B70) : Colors.transparent,
        borderRadius: BorderRadius.circular(9),
      ),
      child: Row(
        children: [
          Icon(
            icon,
            size: 19,
            color: selected ? Colors.white : const Color(0xFFB7C4E3),
          ),
          const SizedBox(width: 12),
          Text(
            title,
            style: TextStyle(
              color: selected ? Colors.white : const Color(0xFFB7C4E3),
              fontWeight: selected ? FontWeight.w600 : FontWeight.normal,
              fontSize: 13,
            ),
          ),
        ],
      ),
    );
  }

  void saveJob() {
    if (!_formKey.currentState!.validate()) return;

    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(
        content: Text(
          'Form completed! Firestore saving will be connected next.',
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: background,
      body: LayoutBuilder(
        builder: (context, constraints) {
          final isDesktop = constraints.maxWidth >= 850;

          return Row(
            children: [
              if (isDesktop)
                Container(
                  width: 235,
                  color: navy,
                  padding: const EdgeInsets.all(20),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      const SizedBox(height: 15),
                      const Row(
                        children: [
                          Icon(
                            Icons.work_outline,
                            color: Color(0xFF8FA5FF),
                            size: 28,
                          ),
                          SizedBox(width: 10),
                          Text(
                            'HireFlow',
                            style: TextStyle(
                              color: Colors.white,
                              fontWeight: FontWeight.bold,
                              fontSize: 21,
                            ),
                          ),
                        ],
                      ),
                      const SizedBox(height: 45),
                      const Text(
                        'WORKSPACE',
                        style: TextStyle(
                          color: Color(0xFF8293B8),
                          fontSize: 10,
                          letterSpacing: 1.5,
                          fontWeight: FontWeight.bold,
                        ),
                      ),
                      const SizedBox(height: 17),
                      sidebarItem(Icons.grid_view_rounded, 'Overview'),
                      sidebarItem(
                        Icons.add_box_outlined,
                        'Create Job',
                        selected: true,
                      ),
                      sidebarItem(Icons.work_outline, 'Manage Jobs'),
                      sidebarItem(Icons.people_outline, 'Candidates'),
                      sidebarItem(Icons.analytics_outlined, 'Analytics'),
                      const Spacer(),
                      const Divider(color: Color(0xFF35456E)),
                      sidebarItem(Icons.settings_outlined, 'Settings'),
                      sidebarItem(Icons.logout, 'Logout'),
                    ],
                  ),
                ),
              Expanded(
                child: Column(
                  children: [
                    Container(
                      height: 75,
                      padding: const EdgeInsets.symmetric(horizontal: 25),
                      decoration: const BoxDecoration(
                        color: Colors.white,
                        border: Border(
                          bottom: BorderSide(color: Color(0xFFE9EDF5)),
                        ),
                      ),
                      child: Row(
                        children: [
                          if (!isDesktop)
                            Icon(Icons.work_outline, color: blue, size: 25),
                          if (!isDesktop) const SizedBox(width: 10),
                          const Expanded(
                            child: Text(
                              'Recruiter Workspace',
                              style: TextStyle(
                                fontWeight: FontWeight.bold,
                                fontSize: 16,
                                color: Color(0xFF172554),
                              ),
                            ),
                          ),
                          const Icon(
                            Icons.notifications_none,
                            color: Color(0xFF64748B),
                          ),
                          const SizedBox(width: 18),
                          CircleAvatar(
                            radius: 18,
                            backgroundColor: const Color(0xFFEEF2FF),
                            child: Icon(Icons.person, color: blue, size: 20),
                          ),
                          const SizedBox(width: 9),
                          if (constraints.maxWidth > 500)
                            const Text(
                              'Administrator',
                              style: TextStyle(
                                fontSize: 12,
                                fontWeight: FontWeight.w600,
                              ),
                            ),
                        ],
                      ),
                    ),
                    Expanded(
                      child: SingleChildScrollView(
                        padding: const EdgeInsets.all(24),
                        child: Center(
                          child: ConstrainedBox(
                            constraints: const BoxConstraints(maxWidth: 950),
                            child: Form(
                              key: _formKey,
                              child: Column(
                                crossAxisAlignment: CrossAxisAlignment.start,
                                children: [
                                  const Text(
                                    'Create a New Job',
                                    style: TextStyle(
                                      fontSize: 27,
                                      fontWeight: FontWeight.bold,
                                      color: Color(0xFF172554),
                                    ),
                                  ),
                                  const SizedBox(height: 7),
                                  const Text(
                                    'Define the role and qualifications to find your ideal candidate.',
                                    style: TextStyle(
                                      fontSize: 13,
                                      color: Color(0xFF64748B),
                                    ),
                                  ),
                                  const SizedBox(height: 25),

                                  sectionCard(
                                    title: 'Job Information',
                                    subtitle: 'Basic details about this position',
                                    icon: Icons.work_outline,
                                    children: [
                                      inputField(
                                        label: 'Job Title',
                                        hint: 'e.g. Junior AI Engineer',
                                        controller: _titleController,
                                      ),
                                      const SizedBox(height: 18),
                                      LayoutBuilder(
                                        builder: (context, box) {
                                          if (box.maxWidth < 550) {
                                            return Column(
                                              children: [
                                                inputField(
                                                  label: 'Department',
                                                  hint: 'e.g. Artificial Intelligence',
                                                  controller: _departmentController,
                                                ),
                                                const SizedBox(height: 18),
                                                inputField(
                                                  label: 'Job Location',
                                                  hint: 'e.g. Lahore, Pakistan',
                                                  controller: _locationController,
                                                ),
                                              ],
                                            );
                                          }
                                          return Row(
                                            children: [
                                              Expanded(
                                                child: inputField(
                                                  label: 'Department',
                                                  hint: 'e.g. Artificial Intelligence',
                                                  controller: _departmentController,
                                                ),
                                              ),
                                              const SizedBox(width: 15),
                                              Expanded(
                                                child: inputField(
                                                  label: 'Job Location',
                                                  hint: 'e.g. Lahore, Pakistan',
                                                  controller: _locationController,
                                                ),
                                              ),
                                            ],
                                          );
                                        },
                                      ),
                                      const SizedBox(height: 18),
                                      LayoutBuilder(
                                        builder: (context, box) {
                                          if (box.maxWidth < 550) {
                                            return Column(
                                              children: [
                                                dropdownField(
                                                  label: 'Employment Type',
                                                  value: _jobType,
                                                  items: [
                                                    'Full Time',
                                                    'Part Time',
                                                    'Internship',
                                                    'Contract',
                                                    'Remote',
                                                  ],
                                                  onChanged: (v) => setState(
                                                    () => _jobType = v!,
                                                  ),
                                                ),
                                                const SizedBox(height: 18),
                                                dropdownField(
                                                  label: 'Experience Required',
                                                  value: _experience,
                                                  items: [
                                                    'Fresh Graduate',
                                                    'Less than 1 Year',
                                                    '1-2 Years',
                                                    '3-5 Years',
                                                    '5+ Years',
                                                  ],
                                                  onChanged: (v) => setState(
                                                    () => _experience = v!,
                                                  ),
                                                ),
                                              ],
                                            );
                                          }
                                          return Row(
                                            children: [
                                              Expanded(
                                                child: dropdownField(
                                                  label: 'Employment Type',
                                                  value: _jobType,
                                                  items: [
                                                    'Full Time',
                                                    'Part Time',
                                                    'Internship',
                                                    'Contract',
                                                    'Remote',
                                                  ],
                                                  onChanged: (v) => setState(
                                                    () => _jobType = v!,
                                                  ),
                                                ),
                                              ),
                                              const SizedBox(width: 15),
                                              Expanded(
                                                child: dropdownField(
                                                  label: 'Experience Required',
                                                  value: _experience,
                                                  items: [
                                                    'Fresh Graduate',
                                                    'Less than 1 Year',
                                                    '1-2 Years',
                                                    '3-5 Years',
                                                    '5+ Years',
                                                  ],
                                                  onChanged: (v) => setState(
                                                    () => _experience = v!,
                                                  ),
                                                ),
                                              ),
                                            ],
                                          );
                                        },
                                      ),
                                    ],
                                  ),
                                  const SizedBox(height: 18),

                                  sectionCard(
                                    title: 'Job Description',
                                    subtitle: 'Explain the position to applicants',
                                    icon: Icons.description_outlined,
                                    children: [
                                      inputField(
                                        label: 'Job Overview',
                                        hint: 'Describe the role, goals and what the candidate will do...',
                                        controller: _descriptionController,
                                        maxLines: 5,
                                      ),
                                      const SizedBox(height: 18),
                                      inputField(
                                        label: 'Responsibilities',
                                        hint: 'List the main responsibilities of this position...',
                                        controller: _responsibilitiesController,
                                        maxLines: 4,
                                      ),
                                    ],
                                  ),
                                  const SizedBox(height: 18),

                                  sectionCard(
                                    title: 'Candidate Requirements',
                                    subtitle: 'Set the qualifications for your ideal candidate',
                                    icon: Icons.school_outlined,
                                    children: [
                                      dropdownField(
                                        label: 'Minimum Education',
                                        value: _education,
                                        items: [
                                          'Intermediate',
                                          'Diploma',
                                          "Bachelor's Degree",
                                          "Master's Degree",
                                          'PhD',
                                          'Any Education Level',
                                        ],
                                        onChanged: (v) => setState(
                                          () => _education = v!,
                                        ),
                                      ),
                                      const SizedBox(height: 18),
                                      inputField(
                                        label: 'Required Skills',
                                        hint: 'e.g. Python, Flutter, Machine Learning, SQL',
                                        controller: _skillsController,
                                        maxLines: 3,
                                      ),
                                      const SizedBox(height: 8),
                                      const Text(
                                        'Separate skills with commas.',
                                        style: TextStyle(
                                          color: Color(0xFF94A3B8),
                                          fontSize: 11,
                                        ),
                                      ),
                                    ],
                                  ),
                                  const SizedBox(height: 25),

                                  Row(
                                    children: [
                                      Expanded(
                                        child: OutlinedButton(
                                          onPressed: () {
                                            _formKey.currentState?.reset();
                                            _titleController.clear();
                                            _departmentController.clear();
                                            _locationController.clear();
                                            _descriptionController.clear();
                                            _responsibilitiesController.clear();
                                            _skillsController.clear();
                                            setState(() {
                                              _jobType = 'Full Time';
                                              _experience = '1-2 Years';
                                              _education = "Bachelor's Degree";
                                            });
                                          },
                                          style: OutlinedButton.styleFrom(
                                            padding: const EdgeInsets.symmetric(
                                              vertical: 17,
                                            ),
                                            foregroundColor: navy,
                                            side: const BorderSide(
                                              color: Color(0xFFCBD5E1),
                                            ),
                                            shape: RoundedRectangleBorder(
                                              borderRadius: BorderRadius.circular(10),
                                            ),
                                          ),
                                          child: const Text('Clear Form'),
                                        ),
                                      ),
                                      const SizedBox(width: 14),
                                      Expanded(
                                        child: ElevatedButton.icon(
                                          onPressed: saveJob,
                                          icon: const Icon(Icons.add, size: 19),
                                          label: const Text('Create Job'),
                                          style: ElevatedButton.styleFrom(
                                            backgroundColor: blue,
                                            foregroundColor: Colors.white,
                                            elevation: 0,
                                            padding: const EdgeInsets.symmetric(
                                              vertical: 17,
                                            ),
                                            shape: RoundedRectangleBorder(
                                              borderRadius: BorderRadius.circular(10),
                                            ),
                                          ),
                                        ),
                                      ),
                                    ],
                                  ),
                                  const SizedBox(height: 30),
                                ],
                              ),
                            ),
                          ),
                        ),
                      ),
                    ),
                  ],
                ),
              ),
            ],
          );
        },
      ),
    );
  }
}
